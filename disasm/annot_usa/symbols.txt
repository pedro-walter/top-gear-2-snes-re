# Routine names and documentation (USA). "@BBAAAA name" then ";; text" lines (first line = one-line summary).
# Confidence: no mark = verified (code read and/or emulator), (?) = inferred.  Addresses are LoROM bank:offset (banks 00..1F).

# ---------------------------------------------------------------- boot
@008A38 reset_vec
;; RESET vector: CLC; XCE; JML 01:8003 (native mode, then the 01:8008 init sequence).
@018008 boot_init
;; Forced blank, wait two vblanks, DP=0, S=$07FF, JSR 01F7FD (title/menu data setup) then JML 1F:8000 (front-end).
@1F8000 frontend_entry
;; Front-end (title / menus / car shop) entry: DBR=$9F, S=$1FFF, DP=$0100.  All menu code lives in bank 1F; it busy-waits on $4211/$4212.

# ---------------------------------------------------------------- interrupts
@01E2F2 nmi_handler
;; NMI body the WRAM stub (7E0000: JML $81E2F2) jumps to.  Saves regs, DP=0, DBR=$82, copies $0122->$22, fades the screen
;; brightness in $58/$5D (fixed-point 8.4 steps, limit $0F00) and bumps frame counters $1A/$1C.
# ---------------------------------------------------------------- decompression
@01DC53 decomp_to_wram
;; LZ/RLE decompressor to WRAM.  A=src bank (low byte) | dest-bank bit in A.high bit0 (=> 7E/7F), X=src offset, Y=dest offset.
;; Writes through WMDATA ($2180).  Command byte: top 3 bits select one of 7 handlers via the jump table at 01:DC91
;; (literal run / long literal / RLE / long RLE / back-reference variants); cmd $00 ends the stream.  Returns A = bytes written.
@01DE85 decomp_to_wram_b
;; Second decompressor variant (17 callers), same register convention, its own handler table at 01:DEC2.

# ---------------------------------------------------------------- race frame
@008A41 player_car_update
;; Per-frame PLAYER car update (DP=$0600): physics (accel curve 80:BE33 by engine level, gear ratios 80:BDFB, drag), position along the track, collisions.  docs/03-physics.md
@00D726 car_list_update
;; Updates the car/object records (see ram.txt: 0x40-byte records at 7E:C400...).
@00D84B car_dispatch_0
;; Start of a run of 20 unrolled blocks that copy record fields and call the per-record handler `JSR ($E417,X)`.
@00F042 mul_16x8
;; Small signed multiply helper (A * byte -> $01CC/$01CE/$01D0 accumulators via the PPU multiplier).  ~10% of race CPU time.
@08CB74 road_render
;; Biggest per-frame routine (~36% of race CPU time): builds the road scanline tables; uses PPU multiplier and the hardware divider ($4204-$4214).
@0B8000 hdma_build
;; Builds the per-scanline HDMA tables in WRAM (double buffered, selected by sign of $0124: buffers around $AD7E / $AE7E).
@0B8578 road_lines
;; Large per-scanline loop (14% of race CPU time), calls 0BA49B/0BA6EA/0BD7E7.
@0BA6EA road_lines_inner
;; ~11% of race CPU time; heavy use of the PPU multiplier.
@088BCA race_frame_b
;; Called each race frame together with 008A41/00D726/08CB74/0B8000 (frame pipeline at 01:828B).
;; 00:8A41 is really the PLAYER CAR UPDATE (physics + movement), not just road scroll: DP=$0600; speed 24-bit at $061C-$061E,
;; accel from gear tables (80:BE33, 80:C633, 80:CA33), drag from 80:C833 indexed by speed, all via the $211B/$211C multiplier ($2134-36).

# ---------------------------------------------------------------- race session (docs/12-race-session.md)
@018145 race_session_start
;; 1P race start: 00:CD09 (track number), 00:CD5E car setup, 00:D0DA AI setup, first road_render/hdma_build, HDMA programming, then the frame loop.  2P variant at 01:840B.
@018272 race_frame_loop
;; Frame loop: wait $0122, dt=$F8, 01:83BF pause/quit test, 00:8A41 player, 00:D726 cars, 08:8BCA, 08:CB74 render, 0B:8000 HDMA, finish logic, exit via JML [$0076].
@0183BF race_pause_quit
;; Start edge ($0030 bit12) pauses; L+R held while paused returns carry set = quit.
@00CD09 race_track_number
;; Derives the track number ($0108) for the race from the country/race index.
@00CD5E car_setup
;; Turns the shop levels of each player into the physics variables ($068C engine, $0698, $06C0 gears...).  docs/09-car-setup.md
@00D0DA race_ai_setup
;; Per-race setup incl. AI car records (D1D4).  docs/05-ai-cars.md
@00DE9B car_lap_finish
;; Lap counter +0x22 of a car; at $019E laps the car finishes: id -> 7E:FE9E[$06EE], time -> 7E:FEC6, placement for results.
@00E6F8 player_finish_cruise
;; Switches P1 to handler 0x0A (coast to a stop) after finishing.
@01DA29 sound_mailbox
;; Once per frame (DP=$0500): sends up to four 4-byte commands to the SPC700 through ports $2140-43.  docs/08-sound.md
@01E25B rng
;; Random number generator.
# ---------------------------------------------------------------- front-end flow (docs/10-game-flow.md)
@1F809B frontend_reentry

@1F81D4 load_persistent_state
